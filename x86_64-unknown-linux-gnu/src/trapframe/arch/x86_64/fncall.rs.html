1//! Switch context by function call within the same privilege level.
2//!
3//! # Assumption
4//!
5//! This module suppose you are running kernel on Linux or macOS with glibc,
6//! and your user program is based on musl libc.
7//!
8//! Because we will store values in their pthread structure.
9
10use super::{UserContext, UserContextWithExtensions};
11use core::arch::global_asm;
12
13unsafe extern "sysv64" {
14    /// The syscall entry of function call.
15    ///
16    /// # Usage
17    ///
18    /// Replace `syscall` instruction by a `call` instruction.
19    ///
20    /// ```asm
21    /// syscall
22    /// call syscall_fn_entry
23    /// ```
24    pub fn syscall_fn_entry();
25
26    fn syscall_fn_return(regs: &mut UserContext);
27    fn syscall_fn_return_extended(regs: &mut UserContextWithExtensions);
28}
29
30impl UserContext {
31    /// Go to user context by function return, within the same privilege level.
32    ///
33    /// User program should call `syscall_fn_entry()` to return back.
34    /// Trap reason and error code will always be set to 0x100 and 0.
35    pub fn run_fncall(&mut self) {
36        unsafe { syscall_fn_return(self) }
37        self.trap_num = 0x100;
38        self.error_code = 0;
39    }
40}
41
42impl UserContextWithExtensions {
43    /// Goes to user context while preserving x87 and SSE state.
44    pub fn run_fncall(&mut self) {
45        unsafe { syscall_fn_return_extended(self) }
46        self.trap_num = 0x100;
47        self.error_code = 0;
48    }
49}

51// User: (musl)
52// - fs:0  (pthread.self)       = user fsbase
53// - fs:48 (pthread.canary2)    = kernel fsbase
54//
55// Kernel: (glibc)
56// - fs:0  (pthread.self)       = kernel fsbase
57// - fs:64 (pthread.???)        = kernel stack
58// - fs:72 (pthread.???)        = init user fsbase
59//
60#[cfg(target_os = "linux")]
61global_asm!(
62    r#"
63.macro SWITCH_TO_KERNEL_STACK
64    mov rsp, fs:48          # rsp = kernel fsbase
65    mov rsp, [rsp + 64]     # rsp = kernel stack
66.endm
67.macro SAVE_KERNEL_STACK
68    mov fs:64, rsp
69.endm
70.macro PUSH_USER_FSBASE
71    push fs:0
72.endm
73.macro SWITCH_TO_KERNEL_FSBASE
74    mov eax, 158            # SYS_arch_prctl
75    mov edi, 0x1002         # SET_FS
76    mov rsi, fs:48          # rsi = kernel fsbase
77    syscall
78.endm
79.macro POP_USER_FSBASE
80    mov rsi, [rsp + 18 * 8] # rsi = user fsbase
81    mov rdx, fs:0           # rdx = kernel fsbase
82    test rsi, rsi
83    jnz 1f                  # if not 0, goto set
840:  lea rsi, [rdx + 72]     # rsi = init user fsbase
85    mov [rsi], rsi          # user_fs:0 = user fsbase
861:  mov eax, 158            # SYS_arch_prctl
87    mov edi, 0x1002         # SET_FS
88    syscall                 # set fsbase
89    mov fs:48, rdx          # user_fs:48 = kernel fsbase
90.endm
91
92.global syscall_fn_entry
93.global syscall_fn_return
94.global syscall_fn_return_extended
95"#
96);

98// User: (musl)
99// - gs:0   (pthread.self)      = user gsbase
100// - gs:48  (pthread.canary2)   = kernel gsbase
101//
102// Kernel: (darwin)
103// - gs:0   (pthread.tsd[self]) = kernel gsbase - 224
104// - gs:48  (pthread.tsd[6])    = kernel stack
105// - gs:240 (pthread.tsd[30])   = init user fsbase
106//
107// Ref:
108// - Set gsbase:
109//   - https://gist.github.com/aras-p/5389747
110// - Get gsbase:
111//   - https://github.com/DynamoRIO/dynamorio/issues/1568#issuecomment-239819506
112//   - https://github.com/apple/darwin-libpthread/blob/03c4628c8940cca6fd6a82957f683af804f62e7f/src/internal.h#L241
113#[cfg(target_os = "macos")]
114global_asm!(
115    r#"
116.macro SWITCH_TO_KERNEL_STACK
117    mov rsp, gs:48          # rsp = kernel gsbase
118    mov rsp, [rsp + 48]     # rsp = kernel stack
119.endm
120.macro SAVE_KERNEL_STACK
121    mov gs:48, rsp
122.endm
123.macro PUSH_USER_FSBASE
124    push gs:0
125.endm
126.macro SWITCH_TO_KERNEL_FSBASE
127    mov rdi, gs:48          # rdi = kernel gsbase
128    mov eax, 0x3000003
129    syscall                 # set gsbase
130.endm
131.macro POP_USER_FSBASE
132    mov rdi, [rsp + 18 * 8] # rdi = user gsbase
133    mov rsi, gs:0
134    add rsi, 224            # rsi = kernel gsbase
135    test rdi, rdi
136    jnz 1f                  # if not 0, goto set
1370:  lea rdi, [rsi + 30*8]   # rdi = init user gsbase
138                            #     = pthread.tsd[30] (kernel gsbase + 30 * 8)
139    mov [rdi], rdi          # user_gs:0 = user gsbase
1401:  mov eax, 0x3000003
141    syscall                 # set gsbase
142    mov gs:48, rsi          # user_gs:48 = kernel gsbase
143.endm
144
145.global _syscall_fn_entry
146.global syscall_fn_entry
147.global _syscall_fn_return
148.global _syscall_fn_return_extended
149.set _syscall_fn_entry, syscall_fn_entry
150.set _syscall_fn_return, syscall_fn_return
151.set _syscall_fn_return_extended, syscall_fn_return_extended
152"#
153);
154
155global_asm!(
156    r#"
157syscall_fn_entry:
158    pushfq                  # preserve flags before inspecting the tagged context
159    # save rsp
160    lea r11, [rsp + 16]     # save rsp to r11 (clobber)
161
162    SWITCH_TO_KERNEL_STACK
163    pop rsp
164    test rsp, 1
165    jnz 1f
166    and rsp, -2
167    mov qword ptr [rsp + 21*8], 0
168    jmp 2f
1691:  and rsp, -2
170    mov qword ptr [rsp + 21*8], 1
1712:
172    lea rsp, [rsp + 20*8]   # rsp = top of trap frame
173
174    # push trap frame (struct GeneralRegs)
175    push 0                  # ignore gs_base
176    PUSH_USER_FSBASE
177    push [r11 - 16]         # push saved rflags
178    push [r11 - 8]          # push rip
179    push r15
180    push r14
181    push r13
182    push r12
183    push r11
184    push r10
185    push r9
186    push r8
187    push r11                # push rsp
188    push rbp
189    push rdi
190    push rsi
191    push rdx
192    push rcx
193    push rbx
194    push rax
195
196    cmp qword ptr [rsp + 21*8], 0
197    je 1f
198    fxsave64 [rsp + 22*8]
1991:
200
201    # restore callee-saved registers
202    SWITCH_TO_KERNEL_STACK
203    pop rbx
204    pop rbx
205    pop rbp
206    pop r12
207    pop r13
208    pop r14
209    pop r15
210
211    SWITCH_TO_KERNEL_FSBASE
212
213    # go back to Rust
214    ret
215
216    # extern "sysv64" fn syscall_fn_return(&mut UserContext)
217syscall_fn_return:
218    xor esi, esi
219    jmp syscall_fn_return_common
220syscall_fn_return_extended:
221    mov esi, 1
222syscall_fn_return_common:
223    # save callee-saved registers
224    push r15
225    push r14
226    push r13
227    push r12
228    push rbp
229    push rbx
230
231    or rdi, rsi
232    push rdi
233    SAVE_KERNEL_STACK
234    and rdi, -2
235    mov rsp, rdi
236
237    test esi, esi
238    jz 1f
239    fxrstor64 [rsp + 22*8]
2401:
241
242    POP_USER_FSBASE
243
244    # pop trap frame (struct GeneralRegs)
245    pop rax
246    pop rbx
247    pop rcx
248    pop rdx
249    pop rsi
250    pop rdi
251    pop rbp
252    pop r8                  # skip rsp
253    pop r8
254    pop r9
255    pop r10
256    pop r11
257    pop r12
258    pop r13
259    pop r14
260    pop r15
261    pop r11                 # r11 = rip. FIXME: don't overwrite r11!
262    popfq                   # pop rflags
263    mov rsp, [rsp - 8*11]   # restore rsp
264    jmp r11                 # restore rip
265"#
266);

328    #[test]
329    fn run_fncall() {
330        unsafe extern "sysv64" {
331            fn dump_registers();
332        }
333        let mut stack = [0u8; 0x1000];
334        let general = GeneralRegs {
335            rax: 0,
336            rbx: 1,
337            rcx: 2,
338            rdx: 3,
339            rsi: 4,
340            rdi: 5,
341            rbp: 6,
342            rsp: stack.as_mut_ptr() as usize + 0x1000,
343            r8: 8,
344            r9: 9,
345            r10: 10,
346            r11: 11,
347            r12: 12,
348            r13: 13,
349            r14: 14,
350            r15: 15,
351            rip: dump_registers as *const () as usize,
352            rflags: 0,
353            fsbase: 0, // don't set to non-zero garbage value
354            gsbase: 0,
355        };
356        #[repr(C)]
357        struct GuardedContext {
358            context: UserContext,
359            guard: [u8; 512],
360        }
361        let mut legacy = GuardedContext {
362            context: UserContext {
363                general,
364                ..Default::default()
365            },
366            guard: [0xa5; 512],
367        };
368        let mut legacy_xmm0 = [0; 16];
369        legacy.context.run_fncall();
370        unsafe {
371            asm!(
372                "movdqu [{buffer}], xmm0",
373                buffer = in(reg) legacy_xmm0.as_mut_ptr(),
374                options(nostack)
375            );
376        }
377        assert_eq!(legacy.context.trap_num, 0x100);
378        assert_eq!(legacy.guard, [0xa5; 512]);
379        assert_eq!(legacy_xmm0, UPDATED_XMM0);
380
381        let mut cx = UserContextWithExtensions {
382            general,
383            trap_num: 0,
384            error_code: 0,
385            fp_simd: Default::default(),
386        };
387        cx.fp_simd.bytes[160..176].fill(0x5a);
388        cx.run_fncall();
389        let restored_xmm0 = unsafe { core::ptr::addr_of!(RESTORED_XMM0).read_volatile() };
390        assert_eq!(restored_xmm0, [0x5a; 16]);
391        assert_eq!(&cx.fp_simd.bytes[160..176], &UPDATED_XMM0);
392        // check restored registers
393        let general = unsafe { *(cx.general.rsp as *const GeneralRegs) };
394        assert_eq!(
395            general,
396            GeneralRegs {
397                rax: 0,
398                rbx: 1,
399                rcx: 2,
400                rdx: 3,
401                rsi: 4,
402                rdi: 5,
403                rbp: 6,
404                // skip rsp
405                r8: 8,
406                r9: 9,
407                r10: 10,
408                // skip r11
409                r12: 12,
410                r13: 13,
411                r14: 14,
412                r15: 15,
413                ..general
414            }
415        );
416        // check saved registers
417        assert_eq!(
418            cx.general,
419            GeneralRegs {
420                rax: 10,
421                rbx: 11,
422                rcx: 12,
423                rdx: 13,
424                rsi: 14,
425                rdi: 15,
426                rbp: 16,
427                // skip rsp
428                r8: 18,
429                r9: 19,
430                r10: 20,
431                // skip r11
432                r12: 22,
433                r13: 23,
434                r14: 24,
435                r15: 25,
436                ..cx.general
437            }
438        );
439        assert_eq!(cx.trap_num, 0x100);
440        assert_eq!(cx.error_code, 0);
441    }
